1"""Detector: conformance_breach — control-plane state changes the model rejects. 

2 

3This module exposes ``conformance_breach``, which replays the ``state_writes`` 

4trace through ``factory.observability.conformance`` and surfaces every hop the 

5abstract model does not accept. The calling agent decides whether a given 

6finding is a bug in the chain or a legitimate path missing from the model. 

7 

8Two finding kinds, with different meanings: 

9 

10* ``illegal_transition`` — a sanctioned writer produced a state it is not 

11 permitted to produce. The control plane was violated. 

12* ``coverage_breach`` — a writer that is not declared in the model changed a 

13 story's state, or the write could not be attributed at all. This is the 

14 load-bearing one: it means control-plane state moved through a path nobody 

15 documented, so nothing verified it. 

16 

17Neither blocks a merge. The point is that a state change no one recorded stops 

18being invisible. 

19""" 

20 

21from __future__ import annotations 

22 

23from datetime import datetime 

24from pathlib import Path 

25from typing import Any 

26 

27 

28def conformance_breach(*, root: Path, since: datetime | None = None) -> list[dict[str, Any]]: 

29 """Return control-plane state writes that fail the conformance model. 

30 

31 Parameters 

32 ---------- 

33 root: 

34 Factory root directory (the ``state/events/`` parent). 

35 since: 

36 Only consider trace records at or after this timestamp. ``None`` 

37 considers the whole retained stream. 

38 

39 Returns 

40 ------- 

41 list[dict[str, Any]] 

42 One dict per finding, most recent last: 

43 

44 * ``verdict`` — ``"illegal_transition"`` or ``"coverage_breach"`` 

45 * ``story_id`` / ``app`` — the story whose state moved 

46 * ``from_state`` / ``to_state`` — the hop 

47 * ``writer`` — ``module.function`` that performed the write, or 

48 ``"unknown"`` when attribution failed 

49 * ``reason`` — why the model rejected it 

50 * ``ts`` — when the write happened 

51 

52 Returns an empty list when the trace is empty, absent, or fully 

53 conformant — the expected steady state. A non-empty list is always 

54 worth reading: either the chain did something it should not have, or 

55 the model is missing a path that exists in production. 

56 """ 

57 try: 

58 from factory.observability.conformance import check_trace, load_model 

59 from factory.observability.state_trace import read_state_writes 

60 except Exception: # noqa: BLE001 - a detector must never break the L1 cycle 

61 return [] 

62 

63 try: 

64 records = read_state_writes(software_factory_root=root) 

65 except Exception: # noqa: BLE001 

66 return [] 

67 

68 if since is not None: 

69 cutoff = since.isoformat() 

70 # ISO-8601 UTC strings with a fixed offset compare lexicographically in 

71 # timestamp order, which is how every other detector windows a stream. 

72 records = [r for r in records if str(r.get("ts", "")) >= cutoff] 

73 

74 if not records: 

75 return [] 

76 

77 try: 

78 report = check_trace(records, model=load_model()) 

79 except Exception: # noqa: BLE001 - a malformed model must not break L1 

80 return [] 

81 

82 return [finding.as_dict() for finding in report.findings]
