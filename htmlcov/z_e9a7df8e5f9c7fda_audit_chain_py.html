1"""Tamper-evident hash chain over the factory's own event streams. 

2 

3Why 

4--- 

5``state/events/*.ndjson`` is the factory's memory of itself: the FMS reads it to 

6decide what is wrong, and an operator reads it to decide whether the FMS was 

7right. It was append-only best-effort with no sequence, no linkage, and no 

8origin identity — so a truncated file, a hand-edited line, or a row written by a 

9completely different process were all indistinguishable from real history. There 

10was no integrity code anywhere in ``factory/``. 

11 

12Two concrete failures this addresses: 

13 

14* **Test pollution.** A test process and the live daemon wrote structurally 

15 identical rows. Synthetic failures written by the suite were read back by the 

16 L1 watcher as genuine persona failures and escalated for weeks. The 

17 ``FACTORY_STATE_ROOT`` fixture stops new instances; a chain id makes past and 

18 future confusion *detectable* rather than merely unlikely. 

19* **Self-edits.** The factory can rewrite its own code. A verifier the 

20 self-improver could quietly weaken is not a verifier, which is why the tracer 

21 and checker paths are on the L4 forbidden list. 

22 

23Design (from ``buzz-audit``) 

24--------------------------- 

25Each entry carries ``chain_id``, ``seq``, ``prev_hash`` and ``entry_hash``. 

26``chain_id`` is hashed FIRST, so chain identity carries the origin: a row lifted 

27out of one chain can never verify inside another. ``prev_hash`` links entries; 

28``seq`` is monotonic per stream. The payload is canonicalised (sorted keys) 

29before hashing so the digest is stable across processes and Python versions, and 

30optional fields get an explicit presence byte so ``None`` and ``""`` hash 

31differently. 

32 

33Honest about what this is NOT: a local hash chain detects truncation, reordering 

34and in-place edits. It does not defend against an attacker who can rewrite the 

35whole file *and* the head file, because both are writable by the same process. 

36It makes tampering evident, not impossible. 

37"""

51# The 32-byte sentinel hashed in place of ``prev_hash`` for a stream's first 

52# entry. Stored as ``prev_hash = None``; hashed as all-zero bytes. 

53GENESIS_HASH: bytes = b"\x00" * 32 

54 

55# Files that hold chain state, alongside the streams they describe. 

56CHAIN_ID_FILENAME = ".chain_id" 

57CHAIN_HEADS_FILENAME = ".chainheads.json" 

58 

59# Verification verdicts. 

60OK = "ok" 

61UNCHAINED_LEGACY_ROWS = "unchained_legacy_rows" 

62TRUNCATED_BY_ROTATION = "truncated_by_rotation" 

63SEQ_GAP = "seq_gap" 

64BROKEN_LINK = "broken_link" 

65FOREIGN_CHAIN_ID = "foreign_chain_id" 

66CORRUPT_ENTRY = "corrupt_entry" 

67 

68# Verdicts that mean "someone changed history", as opposed to the benign 

69# explanations (rotation dropped a prefix, rows predate the chain). 

70# 

71# SEQ_GAP counts as tampering, which is not obvious. Rotation only ever drops 

72# the OLDEST segment, so the segments that remain are always contiguous at the 

73# end — a gap at the START is truncation (benign, its own verdict), but a gap 

74# MID-STREAM cannot be explained by rotation. It means records between two 

75# retained entries were removed. Excluding it here would have made deletion 

76# undetectable, which a test caught. 

77TAMPER_VERDICTS = frozenset({BROKEN_LINK, FOREIGN_CHAIN_ID, CORRUPT_ENTRY, SEQ_GAP}) 

78 

79# Fields the hash covers, in a FIXED order. Changing this order or set 

80# invalidates every existing chain — which is why it lives in one place. 

81_HASHED_FIELDS = ("ts", "event")

84def canonical_json(value: Any) -> str: 

85 """Serialise with sorted keys so the digest is machine-independent. 

86 

87 A serialisation failure is a hard error, never silently hashed as empty: a 

88 payload we cannot canonicalise is a payload we cannot honestly attest to. 

89 """ 

90 return json.dumps(value, sort_keys=True, separators=(",", ":"), default=repr)

93def _hash_optional(hasher: Any, value: str | None) -> None: 

94 """Feed an optional string with a presence tag. 

95 

96 Without the tag, ``None`` and ``""`` produce the same digest, so a field 

97 could be blanked without breaking the chain. 

98 """ 

99 if value is None: 

100 hasher.update(b"\x00") 

101 else: 

102 hasher.update(b"\x01") 

103 hasher.update(value.encode("utf-8", "replace"))

106def compute_entry_hash( 

107 *, 

108 chain_id: str, 

109 stream: str, 

110 seq: int, 

111 prev_hash: str | None, 

112 record: dict[str, Any], 

113) -> str: 

114 """Return the hex SHA-256 digest binding this entry to its chain. 

115 

116 ``chain_id`` leads the hash (buzz-audit's tenant-binding trick): an entry 

117 cannot be moved between chains and still verify. ``stream`` is included for 

118 the same reason at one level down — a row cannot be lifted from ``runs`` into 

119 ``alerts``. 

120 """ 

121 hasher = hashlib.sha256() 

122 hasher.update(chain_id.encode("utf-8")) 

123 hasher.update(stream.encode("utf-8")) 

124 hasher.update(seq.to_bytes(8, "big")) 

125 for field_name in _HASHED_FIELDS: 

126 raw = record.get(field_name) 

127 _hash_optional(hasher, None if raw is None else str(raw)) 

128 # The payload minus the chain fields themselves — those are structural, and 

129 # including entry_hash would be circular. 

130 payload = { 

131 k: v for k, v in record.items() if k not in ("chain_id", "seq", "prev_hash", "entry_hash") 

132 } 

133 hasher.update(canonical_json(payload).encode("utf-8")) 

134 if prev_hash is None: 

135 hasher.update(GENESIS_HASH) 

136 else: 

137 hasher.update(bytes.fromhex(prev_hash)) 

138 return hasher.hexdigest()

141def chain_id_for(events_dir: Path) -> str: 

142 """Read (or create) the chain identity for ``events_dir``. 

143 

144 One id per event directory. A test run redirected by ``FACTORY_STATE_ROOT`` 

145 gets a fresh directory and therefore a fresh id, so test-written rows can 

146 never verify inside the production chain — the direct fix for the 

147 test-pollution class described in the module docstring. 

148 

149 Best-effort: on any I/O failure the caller writes an unchained record rather 

150 than losing the event. 

151 """ 

152 path = events_dir / CHAIN_ID_FILENAME 

153 try: 

154 existing = path.read_text(encoding="utf-8").strip() 

155 if existing: 

156 return existing 

157 except OSError: 

158 pass 

159 new_id = str(uuid.uuid4()) 

160 try: 

161 events_dir.mkdir(parents=True, exist_ok=True) 

162 # Exclusive create so two racing processes cannot both claim to have 

163 # founded the chain; the loser reads the winner's id. 

164 fd = os.open(str(path), os.O_CREAT | os.O_EXCL | os.O_WRONLY, 0o644) 

165 try: 

166 os.write(fd, (new_id + "\n").encode("utf-8")) 

167 finally: 

168 os.close(fd) 

169 return new_id 

170 except FileExistsError: 

171 # Another writer won the create. It may not have written the id yet, so 

172 # an immediate read can come back EMPTY — and returning ``new_id`` there 

173 # would mint a second identity for the same directory, which verification 

174 # then reports as FOREIGN_CHAIN_ID. Retry briefly instead of inventing 

175 # one. ``append_chained`` also calls this inside its lock, so this path 

176 # is only reachable via a direct caller (the CLI, the verifier). 

177 for _ in range(20): 

178 try: 

179 existing = path.read_text(encoding="utf-8").strip() 

180 except OSError: 

181 break 

182 if existing: 

183 return existing 

184 time.sleep(0.01) 

185 return new_id 

186 except OSError: 

187 return new_id

190def append_chained( 

191 events_dir: Path, 

192 stream: str, 

193 record: dict[str, Any], 

194 write_line: Any, 

195) -> bool: 

196 """Stamp chain fields onto ``record``, append it, and advance the head — 

197 all while holding one exclusive lock. 

198 

199 The lock MUST span reserve → hash → append → commit. An earlier version 

200 reserved the sequence under a lock and committed the head afterwards; with 

201 the factory's tick process and manager daemon writing the same streams 

202 concurrently, both readers saw the same head and were handed the same 

203 ``seq``. A concurrency test caught it producing duplicate sequence numbers 

204 (``[1, 1, 2, 2, 2, 2, ...]`` for 40 writers), which would read as tampering 

205 for a reason that is not tampering. Serialising the whole critical section 

206 also makes the NDJSON append atomic with respect to other writers. 

207 

208 ``write_line`` is called with the finished JSON line while the lock is held; 

209 it must do nothing but append. 

210 

211 Returns ``True`` when the record was written chained, ``False`` when the 

212 chain could not be used (no ``fcntl``, unwritable head file) — in which case 

213 NOTHING has been written and the caller must fall back to an unchained 

214 append. Losing an event is never acceptable; losing its link is. 

215 """ 

216 try: 

217 import fcntl 

218 except ImportError: # pragma: no cover - POSIX-only factory 

219 return False 

220 

221 try: 

222 events_dir.mkdir(parents=True, exist_ok=True) 

223 heads_path = events_dir / CHAIN_HEADS_FILENAME 

224 # "a+" specifically: it creates the file if absent but NEVER truncates. 

225 # An earlier version used "w+" for the create case, which truncates at 

226 # OPEN time — i.e. before the lock is acquired. A second process 

227 # arriving while the first held the lock would blank the head it had 

228 # just written, and the next writer would reuse a sequence number, 

229 # producing a SEQ_GAP that reads as tampering when nothing was tampered 

230 # with. In append mode all writes land at EOF, so the truncate() inside 

231 # the lock below is what makes the rewrite work. 

232 with open(heads_path, "a+", encoding="utf-8") as handle: 

233 fcntl.flock(handle.fileno(), fcntl.LOCK_EX) 

234 try: 

235 # Resolve the chain id INSIDE the lock. Outside it, two writers 

236 # racing on a fresh events dir could each mint a different id: 

237 # the loser of the O_EXCL create can read the file back before 

238 # the winner has written to it, see empty, and fall back to its 

239 # own uuid. Two ids in one stream is reported (correctly) as 

240 # FOREIGN_CHAIN_ID, so the symptom was a rare "tampering" 

241 # verdict on a stream nobody had touched — a ~1-in-10 flake in 

242 # the concurrency test. 

243 chain_id = chain_id_for(events_dir) 

244 handle.seek(0) 

245 raw = handle.read().strip() 

246 heads: dict[str, Any] = json.loads(raw) if raw else {} 

247 if not isinstance(heads, dict): 

248 heads = {} 

249 entry = heads.get(stream) or {} 

250 seq = int(entry.get("seq", 0)) + 1 

251 stored_prev = entry.get("hash") 

252 prev_hash = str(stored_prev) if stored_prev else None 

253 

254 record["chain_id"] = chain_id 

255 record["seq"] = seq 

256 record["prev_hash"] = prev_hash 

257 record["entry_hash"] = compute_entry_hash( 

258 chain_id=chain_id, 

259 stream=stream, 

260 seq=seq, 

261 prev_hash=prev_hash, 

262 record=record, 

263 ) 

264 

265 # Append BEFORE advancing the head: if the append fails, the 

266 # head is untouched and the next writer reuses this sequence. 

267 # The reverse order would leave a permanent gap that reads as a 

268 # deletion. 

269 write_line(json.dumps(record) + "\n") 

270 

271 heads[stream] = {"seq": seq, "hash": record["entry_hash"]} 

272 handle.seek(0) 

273 handle.truncate() 

274 handle.write(json.dumps(heads, sort_keys=True)) 

275 # flush() only — deliberately NOT fsync(). This runs on EVERY 

276 # telemetry write (the live `prompts` stream alone has 45k 

277 # records), and an fsync there costs milliseconds each. flush() 

278 # already makes the head visible to other processes and survives 

279 # a process crash, which is the failure this needs to tolerate. 

280 # A machine crash would lose the appended event line too — that 

281 # is not fsynced either — so fsyncing only the head would be 

282 # both slower and inconsistent. 

283 handle.flush() 

284 return True 

285 finally: 

286 fcntl.flock(handle.fileno(), fcntl.LOCK_UN) 

287 except Exception as exc: # noqa: BLE001 - telemetry path; degrade to unchained 

288 print( 

289 f"[audit_chain] could not chain stream={stream!r} ({exc}); writing unchained", 

290 file=sys.stderr, 

291 ) 

292 for key in ("chain_id", "seq", "prev_hash", "entry_hash"): 

293 record.pop(key, None) 

294 return False

302@dataclass 

303class VerifyReport: 

304 """Result of verifying one stream's chain.""" 

305 

306 stream: str 

307 total_records: int = 0 

308 chained_records: int = 0 

309 unchained_records: int = 0 

310 verdicts: list[str] = field(default_factory=list) 

311 problems: list[dict[str, Any]] = field(default_factory=list) 

312 chain_ids_seen: list[str] = field(default_factory=list) 

313 

314 @property 

315 def tampered(self) -> bool: 

316 """True only for verdicts that mean history was altered. 

317 

318 Rotation truncation and legacy rows are NOT tampering — conflating them 

319 would make the check cry wolf on every real deployment and get it 

320 switched off. 

321 """ 

322 return any(v in TAMPER_VERDICTS for v in self.verdicts) 

323 

324 @property 

325 def verdict(self) -> str: 

326 if self.verdicts: 

327 return self.verdicts[0] 

328 return OK 

329 

330 def as_dict(self) -> dict[str, Any]: 

331 return { 

332 "stream": self.stream, 

333 "total_records": self.total_records, 

334 "chained_records": self.chained_records, 

335 "unchained_records": self.unchained_records, 

336 "verdict": self.verdict, 

337 "verdicts": list(self.verdicts), 

338 "tampered": self.tampered, 

339 "chain_ids_seen": list(self.chain_ids_seen), 

340 "problems": list(self.problems), 

341 }

344def _ordered_segments(events_dir: Path, stream: str) -> list[Path]: 

345 """Stream files oldest-first: rotated ``.N`` (highest N oldest), then live. 

346 

347 Same ordering as ``factory.chain.step_events._ordered_segments``. Rotation 

348 keeps only a few segments, so a chain read this way legitimately starts 

349 mid-sequence — which is why ``TRUNCATED_BY_ROTATION`` is a distinct verdict. 

350 """ 

351 base = events_dir / f"{stream}.ndjson" 

352 try: 

353 rotated = sorted( 

354 events_dir.glob(f"{stream}.ndjson.*"), 

355 key=lambda p: int(p.suffix.lstrip(".")) if p.suffix.lstrip(".").isdigit() else 0, 

356 reverse=True, 

357 ) 

358 except OSError: 

359 rotated = [] 

360 return [*rotated, base]

363def _iter_records(events_dir: Path, stream: str) -> list[dict[str, Any]]: 

364 out: list[dict[str, Any]] = [] 

365 for segment in _ordered_segments(events_dir, stream): 

366 if not segment.exists(): 

367 continue 

368 try: 

369 with segment.open("r", encoding="utf-8") as handle: 

370 for line in handle: 

371 line = line.strip() 

372 if not line: 

373 continue 

374 try: 

375 record = json.loads(line) 

376 except json.JSONDecodeError: 

377 # A partially-written final line (crash mid-append) must 

378 # not make the whole stream unreadable. 

379 continue 

380 if isinstance(record, dict): 

381 out.append(record) 

382 except OSError: 

383 continue 

384 return out

387def verify_stream( 

388 stream: str, 

389 *, 

390 events_dir: Path, 

391 expected_chain_id: str | None = None, 

392) -> VerifyReport: 

393 """Verify ``stream``'s hash chain. Read-only. 

394 

395 Distinguishes four benign conditions from real tampering: 

396 

397 * ``unchained_legacy_rows`` — records written before chaining existed. They 

398 have no chain fields and MUST verify clean, or every existing deployment 

399 would report as tampered on first run. 

400 * ``truncated_by_rotation`` — the first chained record's ``seq`` is not 1, 

401 so rotation dropped an earlier segment. Expected, not suspicious. 

402 * ``seq_gap`` — a jump mid-stream. Usually a rotation boundary; reported 

403 separately so it can be told apart from a deletion. 

404 * ``foreign_chain_id`` / ``broken_link`` / ``corrupt_entry`` — history was 

405 altered, or a row came from a different origin (e.g. a test run's chain). 

406 """ 

407 report = VerifyReport(stream=stream) 

408 records = _iter_records(events_dir, stream) 

409 report.total_records = len(records) 

410 if not records: 

411 return report 

412 

413 resolved_expected = expected_chain_id 

414 if resolved_expected is None: 

415 path = events_dir / CHAIN_ID_FILENAME 

416 try: 

417 resolved_expected = path.read_text(encoding="utf-8").strip() or None 

418 except OSError: 

419 resolved_expected = None 

420 

421 seen_ids: list[str] = [] 

422 prev_hash: str | None = None 

423 prev_seq: int | None = None 

424 

425 for index, record in enumerate(records): 

426 chain_id = record.get("chain_id") 

427 entry_hash = record.get("entry_hash") 

428 if not chain_id or not entry_hash: 

429 report.unchained_records += 1 

430 continue 

431 

432 report.chained_records += 1 

433 chain_id = str(chain_id) 

434 if chain_id not in seen_ids: 

435 seen_ids.append(chain_id) 

436 

437 if resolved_expected is not None and chain_id != resolved_expected: 

438 report.verdicts.append(FOREIGN_CHAIN_ID) 

439 report.problems.append( 

440 { 

441 "index": index, 

442 "verdict": FOREIGN_CHAIN_ID, 

443 "detail": ( 

444 f"record carries chain_id {chain_id} but this directory's " 

445 f"chain is {resolved_expected} — the row originated " 

446 "elsewhere (a different state root, or a test run)" 

447 ), 

448 "ts": record.get("ts"), 

449 } 

450 ) 

451 continue 

452 

453 try: 

454 seq = int(record.get("seq", 0)) 

455 except (TypeError, ValueError): 

456 report.verdicts.append(CORRUPT_ENTRY) 

457 report.problems.append( 

458 {"index": index, "verdict": CORRUPT_ENTRY, "detail": "seq is not an integer"} 

459 ) 

460 continue 

461 

462 if prev_seq is None: 

463 if seq != 1: 

464 report.verdicts.append(TRUNCATED_BY_ROTATION) 

465 report.problems.append( 

466 { 

467 "index": index, 

468 "verdict": TRUNCATED_BY_ROTATION, 

469 "detail": ( 

470 f"chain starts at seq={seq}; earlier entries were " 

471 "dropped by size-based rotation, which is expected" 

472 ), 

473 } 

474 ) 

475 # The first record we can see cannot have its prev_hash checked 

476 # against anything we hold, so trust its own self-consistency only. 

477 prev_hash = record.get("prev_hash") 

478 else: 

479 # Check the sequence and the link INDEPENDENTLY, not as an elif 

480 # chain: a reordering breaks both, and short-circuiting on the seq 

481 # gap hid the broken link so the verdict under-reported what 

482 # happened. 

483 if seq != prev_seq + 1: 

484 report.verdicts.append(SEQ_GAP) 

485 report.problems.append( 

486 { 

487 "index": index, 

488 "verdict": SEQ_GAP, 

489 "detail": ( 

490 f"seq jumped {prev_seq} -> {seq}; rotation only drops the " 

491 "OLDEST segment, so a mid-stream gap means entries were " 

492 "removed" 

493 ), 

494 "ts": record.get("ts"), 

495 } 

496 ) 

497 if record.get("prev_hash") != prev_hash: 

498 report.verdicts.append(BROKEN_LINK) 

499 report.problems.append( 

500 { 

501 "index": index, 

502 "verdict": BROKEN_LINK, 

503 "detail": ( 

504 f"prev_hash {record.get('prev_hash')!r} does not match the " 

505 f"previous entry's hash {prev_hash!r} — an entry was " 

506 "edited, removed, or reordered" 

507 ), 

508 "ts": record.get("ts"), 

509 } 

510 ) 

511 

512 recomputed = compute_entry_hash( 

513 chain_id=chain_id, 

514 stream=stream, 

515 seq=seq, 

516 prev_hash=record.get("prev_hash"), 

517 record=record, 

518 ) 

519 if recomputed != str(entry_hash): 

520 report.verdicts.append(BROKEN_LINK) 

521 report.problems.append( 

522 { 

523 "index": index, 

524 "verdict": BROKEN_LINK, 

525 "detail": ( 

526 "entry_hash does not match the record's contents — the " 

527 "line was edited in place" 

528 ), 

529 "ts": record.get("ts"), 

530 } 

531 ) 

532 

533 prev_hash = str(entry_hash) 

534 prev_seq = seq 

535 

536 if report.unchained_records and UNCHAINED_LEGACY_ROWS not in report.verdicts: 

537 report.verdicts.append(UNCHAINED_LEGACY_ROWS) 

538 report.chain_ids_seen = seen_ids 

539 # Deduplicate while preserving first-seen order so the summary verdict is 

540 # the most significant one encountered. 

541 report.verdicts = sorted(set(report.verdicts), key=lambda v: (v not in TAMPER_VERDICTS, v)) 

542 return report
