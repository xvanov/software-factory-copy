1"""Structural checks over persona files. 

2 

3Nothing validated persona prompts. They are inputs to every paid model call and 

4the chain's entire behaviour depends on them, but the only constraint anywhere 

5was ``path.exists()``. 

6 

7The check that earns this module's existence is 

8:func:`_check_contract_collisions`. Story 14 failed to converge structurally — 

9dev and reviewer disagreed forever — because a persona prompt used example 

10tokens (``orphan`` / ``unassigned``) for an enumerated field whose real values 

11are something else. Dev could not tell an invented example token from a live 

12contract value, implemented the example, and the reviewer correctly rejected it 

13every round. The model was never the problem; the prompt was. 

14 

15So the collision check does not ask "does this prompt mention a state name" — 

16personas legitimately discuss states all the time. It asks the narrower, 

17answerable question: **near a field whose values are enumerated in code, does 

18this prompt show a value that is not in that enum?** That is precisely the 

19mistake that broke story 14, and it is decidable. 

20 

21Severities 

22---------- 

23``error`` the file cannot be trusted as a prompt (unparseable, empty body, 

24 frontmatter that would leak, a contract collision). 

25``warning`` drift worth knowing about but not worth blocking on (a persona with 

26 no model route, a file nothing dispatches). 

27"""

47# Frontmatter keys the loader understands. Anything else is almost certainly a 

48# typo, and a silently-ignored key is a setting that looks applied but is not. 

49_KNOWN_META_KEYS = frozenset( 

50 {"name", "display_name", "description", "model", "temperature", "max_output_tokens"} 

51) 

52 

53# Enumerated contract fields the personas describe to the model, and their REAL 

54# allowed values. A prompt that demonstrates a value outside these sets is 

55# teaching the model a vocabulary the chain will reject. 

56# 

57# ``state`` is resolved lazily from StoryState so it can never drift from the 

58# enum; the others are small and defined at their use sites. 

59_ENUM_CONTRACTS: dict[str, frozenset[str]] = { 

60 "scope": frozenset({"backend", "frontend", "test"}), 

61 "chain_kind": frozenset({"tdd", "docs"}), 

62} 

63 

64# A token only counts as a demonstrated VALUE when it sits in an actual 

65# key-value position: ``"scope": "backend"``, ``scope: backend``, 

66# ``scope="backend"``. Requiring the separator is what makes this check usable. 

67# 

68# An earlier version scanned a 120-character window after any mention of the 

69# field name, which produced 80 findings against the real 22-persona corpus and 

70# every single one was a false positive — it was reading sibling JSON keys 

71# (``"scope": "backend", "rationale": "..."`` flagged ``rationale``) and 

72# ordinary prose as values. A check that noisy is one nobody runs twice. 

73_FIELD_VALUE = r"""["'`]?\b{field}\b["'`]?\s*[:=]\s*["'`]([a-z][a-z0-9_]{{2,40}})["'`]""" 

74 

75# Values that are obviously placeholders rather than claims about the contract. 

76_PLACEHOLDER_VALUES = frozenset( 

77 {"value", "string", "str", "text", "name", "one_of", "enum", "example", "todo", "xxx"} 

78)

129def _check_structure(persona: Persona) -> list[Finding]: 

130 out: list[Finding] = [] 

131 body = persona.body.strip() 

132 if not body: 

133 out.append( 

134 Finding( 

135 persona.name, 

136 ERROR, 

137 "empty_body", 

138 "persona body is empty — the model would be prompted with nothing", 

139 ) 

140 ) 

141 return out 

142 

143 first_line = body.splitlines()[0].strip() 

144 if not first_line.startswith("# "): 

145 out.append( 

146 Finding( 

147 persona.name, 

148 ERROR, 

149 "missing_heading", 

150 f"first line must be an H1 identifying the persona, got {first_line[:60]!r}", 

151 ) 

152 ) 

153 elif "persona" not in first_line.lower(): 

154 out.append( 

155 Finding( 

156 persona.name, 

157 WARNING, 

158 "heading_shape", 

159 f"H1 does not say 'persona': {first_line[:60]!r}", 

160 ) 

161 ) 

162 elif persona.name not in first_line: 

163 # The heading is what a reader (and the model) uses to know which 

164 # persona it is. A heading naming a DIFFERENT persona than the filename 

165 # is how a copy-pasted prompt ends up dispatched under the wrong key. 

166 out.append( 

167 Finding( 

168 persona.name, 

169 ERROR, 

170 "heading_name_mismatch", 

171 f"H1 does not mention the persona key {persona.name!r}: {first_line[:60]!r}", 

172 ) 

173 ) 

174 

175 if len(persona.body.encode("utf-8")) > MAX_BODY_BYTES: 

176 out.append(Finding(persona.name, ERROR, "body_too_large", "body exceeds the size bound")) 

177 return out

180def _check_frontmatter(persona: Persona) -> list[Finding]: 

181 out: list[Finding] = [] 

182 meta = persona.meta 

183 if meta.name and meta.name != persona.name: 

184 out.append( 

185 Finding( 

186 persona.name, 

187 ERROR, 

188 "name_mismatch", 

189 f"frontmatter name {meta.name!r} does not match filename {persona.name!r}", 

190 ) 

191 ) 

192 for key in sorted(meta.extra): 

193 out.append( 

194 Finding( 

195 persona.name, 

196 WARNING, 

197 "unknown_meta_key", 

198 f"frontmatter key {key!r} is not understood and has no effect", 

199 ) 

200 ) 

201 if meta.temperature is not None and not 0.0 <= meta.temperature <= 2.0: 

202 out.append( 

203 Finding( 

204 persona.name, 

205 ERROR, 

206 "temperature_out_of_range", 

207 f"temperature {meta.temperature} is outside 0.0–2.0", 

208 ) 

209 ) 

210 if meta.max_output_tokens is not None and meta.max_output_tokens <= 0: 

211 out.append( 

212 Finding( 

213 persona.name, 

214 ERROR, 

215 "max_output_tokens_invalid", 

216 f"max_output_tokens must be positive, got {meta.max_output_tokens}", 

217 ) 

218 ) 

219 # Frontmatter must never reach the model. 

220 if persona.body.lstrip().startswith("---"): 

221 out.append( 

222 Finding( 

223 persona.name, 

224 ERROR, 

225 "frontmatter_leak", 

226 "body still begins with '---' — frontmatter was not stripped", 

227 ) 

228 ) 

229 return out

232def _check_contract_collisions(persona: Persona) -> list[Finding]: 

233 """Flag demonstrated values that are not in the field's real enum. 

234 

235 The story-14 failure mode: a prompt shows ``"scope": "orphan"`` as an 

236 example, dev implements ``orphan``, the reviewer rejects it because the 

237 contract only allows backend/frontend/test, and the two never converge. The 

238 model behaved correctly on both sides. 

239 

240 Only tokens in an explicit ``field: value`` position count, so prose that 

241 merely mentions a field is never flagged. 

242 """ 

243 out: list[Finding] = [] 

244 contracts = dict(_ENUM_CONTRACTS) 

245 contracts["state"] = _state_values() 

246 body = persona.body 

247 seen: set[tuple[str, str]] = set() 

248 

249 for field_name, allowed in contracts.items(): 

250 pattern = re.compile(_FIELD_VALUE.format(field=re.escape(field_name))) 

251 for match in pattern.finditer(body): 

252 token = match.group(1) 

253 if token in allowed or token in _PLACEHOLDER_VALUES: 

254 continue 

255 if (field_name, token) in seen: 

256 continue 

257 seen.add((field_name, token)) 

258 out.append( 

259 Finding( 

260 persona.name, 

261 ERROR, 

262 "contract_collision", 

263 f"shows {token!r} as a value for {field_name!r}, but the " 

264 f"contract only allows {sorted(allowed)}. A prompt example the " 

265 "chain will reject causes structural dev/reviewer " 

266 "non-convergence (story 14).", 

267 ) 

268 ) 

269 return out

272def _check_routing(persona: Persona) -> list[Finding]: 

273 out: list[Finding] = [] 

274 try: 

275 from factory.model_router import all_known_personas 

276 

277 known = set(all_known_personas()) 

278 except Exception: # noqa: BLE001 - a routing read must not fail validation 

279 return out 

280 

281 if persona.name not in known: 

282 out.append( 

283 Finding( 

284 persona.name, 

285 WARNING, 

286 "no_model_route", 

287 "no entry in routes.yaml — this persona silently uses the block " 

288 "fallback model, so its cost and capability are unintentional", 

289 ) 

290 ) 

291 return out

294def validate_persona(name: str, *, personas_dir: Path | None = None) -> list[Finding]: 

295 """Run every check against one persona.""" 

296 try: 

297 persona = load_persona(name, personas_dir=personas_dir) 

298 except PersonaError as exc: 

299 return [Finding(name, ERROR, "unloadable", str(exc))] 

300 

301 return [ 

302 *_check_structure(persona), 

303 *_check_frontmatter(persona), 

304 *_check_contract_collisions(persona), 

305 *_check_routing(persona),

309def validate_all(*, personas_dir: Path | None = None) -> ValidationReport: 

310 """Validate every persona file present.""" 

311 report = ValidationReport() 

312 for name in available_personas(personas_dir=personas_dir): 

313 report.checked += 1 

314 report.findings.extend(validate_persona(name, personas_dir=personas_dir)) 

315 return report
